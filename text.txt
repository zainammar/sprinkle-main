<script>
function copyTextA() {
const text = document.getElementById("textToCopyA");
text.select();
text.setSelectionRange(0, 99999); // For mobile devices

navigator.clipboard.writeText(text.value).then(() => {
  alert("Copied to clipboard!");
}).catch(err => {
  console.error('Failed to copy: ', err);
});
}

function changeBgColor() {
let color = document.getElementById("bgColorPicker").value;
document.body.style.backgroundColor = color;
}

</script>

<script>
window.addEventListener('DOMContentLoaded', () => {
  // You can change the file names here
  const filesToLoad = {
    content1: 'page1.html',
    content2: 'page2.html'
  };

  // Loop through and load each file into its matching div
  for (const [divId, filename] of Object.entries(filesToLoad)) {
    fetch(filename)
      .then(response => {
        if (!response.ok) throw new Error(`File not found: ${filename}`);
        return response.text();
      })
      .then(data => {
        document.getElementById(divId).innerHTML = data;
      })
      .catch(error => {
        document.getElementById(divId).innerHTML = `<p style="color:red">${error.message}</p>`;
      });
  }
});
</script>
